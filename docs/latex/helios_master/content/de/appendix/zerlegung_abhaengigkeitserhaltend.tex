\chapter{Jede entitätsbezogene Zerlegung $D$ von $K$ ist abhängigkeitserhaltend.}

\begin{lemma}\label{lem:k_abhaengigkeitserhalten}
    Jede entitätsbezogene Zerlegung $D$ von $K$ ist abhängigkeitserhaltend.
\end{lemma}
\begin{proof}
    Zu zeigen ist, dass die Vereinigung der Projektionen von $F_{K}$ auf jede $R_{i} \in D$ äquivalent zu $F_{K}$ ist:
    \[
        (\pi_{R_{1}}(F_{K}) \cup \pi_{R_{2}}(F_{K})  \cup \ldots \cup \pi_{R_{m}}(F_{K}))^+ = (F_{K})^+
    \]
    \smallskip

    \noindent
    Für jede funktionale Abhängigkeitsmenge $F_{R_{i}}$ einer Relation $R_{i} \in D, i \leq m$, gilt:

    \[
        F_{R_{i}} = \{e \rightarrow k_x \mid k_x \in R_{i} \setminus \{e\} \}
    \]
    \medskip
    \noindent
    Daraus können wir unmittelbar
    \[
        (\pi_{R_{i}}(F_{K}))^+ = (F_{R_{i}})^+
    \]
    \noindent
    ableiten~\cite[S. 339--340]{EN11}.\\

    \noindent
    Wegen
    \[
        \bigcup_{i=1}^{m} R_{i} = K
    \]
    sind alle Komponentenattribute aus $K$ entlang den Zerlegungsrelationen verteilt, womit keine funktionale Abhängigkeit mit einer Zerlegung $D$ von $K$ verlorengeht.
    Damit gilt insgesamt

    \[
        \left(\bigcup_{i=1}^m F_{R_{i}}\right)^+ = (F_{K})^+
    \]
\end{proof}
\bigskip
